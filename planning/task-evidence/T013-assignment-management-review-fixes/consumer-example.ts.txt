import assert from 'node:assert/strict';
import { createPracticeAssignmentManager } from './skills/practice-assignment-manager-client';
import { managementContext as context, managementAssignmentId as id } from './fixture';
let reads = 0, writes = 0;
const c = createPracticeAssignmentManager(context, {
 write: async () => {writes++; return {status:404,body:null};},
 read: async () => {reads++; return {status:200,body:context};},
}, () => 'synthetic-consumer-key');
const original = c.snapshot().context;
await c.revoke(id, 'Synthetic missing target.');
assert.equal(c.snapshot().phase, 'STALE');
assert.equal(c.snapshot().context, original);
assert.match(c.snapshot().message, /Refresh current appointments/);
await c.retry();
assert.equal(writes, 1);
await c.load({version:'v1',submissionId:context.page.submissionId,after:null});
assert.equal(reads, 1);
assert.equal(c.snapshot().phase, 'EDITING');
console.log('PASS: independently installed consumer accepts current missing-target recovery contract.');
